# Measurement-error ("smearing") config, for studying the effect of
# different assumed detector resolutions on the analysis. The current
# input ROOT files carry zero measurement error (verified with
# check_smearing.py), so with every section below disabled/zero, the
# analysis sees the exact, unsmeared hits - matching the input files as
# they are today. Edit sigma values and enabled=true to try a resolution
# hypothesis, then run ./run_all.
#
# Loaded/applied by bib_common.load_smearing_config() /
# apply_position_time_smearing() / apply_angle_smearing(). All Gaussian
# (mean 0). Units: mm for position, ns for time, degrees for angles.
#
# Both position and angle errors are specified separately along the
# sensor's own two LOCAL readout axes, u and v (not global X/Y, and not
# the geometric longitudinal/transverse directions used by the
# theta_long_deg/theta_trans_deg diagnostic plots) - a real strip/pixel
# sensor's resolution is generally different along its two readout
# directions (e.g. a strip sensor is far coarser along the strip than
# across it), so u and v each get their own sigma.

# Position smearing acts on the hit's own LOCAL sensor-plane coordinates
# hit_u/hit_v (not global hit_x/hit_y - see the docstring in
# bib_common.apply_position_time_smearing for why); hit_x/y/z are then
# updated consistently via the module's own local axes, and hit_du/
# hit_dv are set to record the sigma actually used.
[position]
sigma_u_mm = 0.100
sigma_v_mm = 0.100
enabled = true

# Time smearing acts on the raw hit_t.
[time]
sigma_t_ns = 0.100
enabled = true

# Angle smearing represents angular-reconstruction uncertainty (e.g.
# from cluster-shape/eta reconstruction, or multiple scattering) along
# the sensor's local u and v axes - see the docstring in
# bib_common.apply_angle_smearing for exactly how. This is the
# MEASURED incidence angle standing in for the (unknowable in a real
# detector) truth momentum direction: once enabled, it drives not only
# the theta_long_deg/theta_trans_deg/theta_full_deg diagnostics but also
# z_axis_intercept_mm and inv_radius_per_mm (hence the momentum_gev
# cut) - all of these are just different projections of the same
# reconstructed hit direction. With sigma=0 (or disabled), the measured
# angle is mathematically identical to the truth angle, so the cuts see
# exactly the unsmeared/truth-based values.
[angle_u]
sigma_deg = 0.5
enabled = true

[angle_v]
sigma_deg = 0.5
enabled = true

# RNG seed, so re-running with the same sigmas reproduces the same
# smeared sample (for a fresh random draw each run, change this between
# runs).
[general]
seed = 42
